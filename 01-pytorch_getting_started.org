from torch import nn

class Landau_model(nn.Module):
    def __init__(self, a0, b, d, T):
        super(Landau_model, self).__init__()
        self.a0 = nn.Parameter(torch.tensor(a0))
        self.b = nn.Parameter(torch.tensor(b))
        self.d = nn.Parameter(torch.tensor(d))
        self.T = nn.Parameter(torch.tensor(T))
        self.Tc = torch.tensor(300.0)

    def forward(self, x):
        yhat = self.a0 * (self.T - self.Tc) * x**2 \
            + self.b * x**4 \
            + self.d * x**6
        return yhat

lm = Landau_model(a0=0.01, b=-1.1, d=1.0, T=300.0)

from torch.utils.data import Dataset, DataLoader

class Data(Dataset):
    def __init__(self, if_outlier=False, seed=0):
        torch.manual_seed(seed)
        x = torch.arange(-1.2, 1.2, 0.01).view(-1, 1) 
        a0 = torch.tensor(0.01)
        b = torch.tensor(-1.1)
        d = torch.tensor(1.0)
        Tc = torch.tensor(300.0)
        T = 320 # Tc + b**2 / (4 * a0 * d)
        model = lambda x: a0 * (T - Tc) * x**2 + b * x**4 + d * x**6
        self.x = x
        self.f = model(x)
        self.y = self.f + 0.05 * torch.rand(self.x.size()) - 0.025 
        self.len = self.x.shape[0]
        if if_outlier:
            self.y[int(self.len * 0.49)] = -0.1
            self.y[int(self.len * 0.5)] = -0.1
            self.y[int(self.len * 0.51)] = -0.1
        
    def __getitem__(self, index):    
        return self.x[index], self.y[index]
    
    def __len__(self):
        return self.len

from torch import nn, optim

def mse(yhat, y):
    return nn.MSELoss()(yhat, y)

def train_model(data, model, epochs, lr=0.1, batch_size=1):
    dataloader = DataLoader(dataset=data, batch_size=batch_size)
    optimizer = optim.SGD(model.parameters(), lr=lr)
    for epoch in range(epochs):
        LOSS = []
        for x, y in dataloader:
            yhat = model(x)
            loss = mse(yhat, y)
            LOSS.append(loss.item())
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
    return LOSS
#+end_src

from torchvision import transforms

class add:
    def __init__(self, i=0, j=0):
        self.i = i
        self.j = j

    def __call__(self, sample):
        x, y = sample
        x += self.i
        y += self.j
        return x, y

class power:
    def __init__(self, j=0):
        self.j = j

    def __call__(self, sample):
        x, y = sample
        y = y**self.j
        return x, y

data1 = OddNum(transform=add(i=0, j=3))
