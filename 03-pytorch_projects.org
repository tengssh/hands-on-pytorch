#+begin_src python :results output :session *python* :exports both
import torch
from torch.utils.data import Dataset, DataLoader
from torch.utils.data import random_split
torch.manual_seed(0)

class Data(Dataset):
    def __init__(self):
        self.x = torch.arange(-1, 1, 0.1).view(-1, 1) 
        self.y = self.x + 1
        self.len = self.x.shape[0]
        
    def __getitem__(self, index):    
        return self.x[index], self.y[index]
    
    def __len__(self):
        return self.len

data = Data()
lengths = [int(data.len * _) for _ in (0.5, 0.3, 0.2)]
traning_set, validation_set, testing_set = random_split(dataset=data, lengths=lengths)
for _ in [traning_set, validation_set, testing_set]:
    print(len(_))
