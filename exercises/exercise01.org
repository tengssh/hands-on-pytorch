import torch
from torch.utils.data import Dataset, DataLoader
from sklearn.preprocessing import MinMaxScaler

class Sunspot_Number(Dataset):
    def __init__(self, df=None, transform=None):
        self._year = torch.tensor(df["Year"].to_numpy().reshape(-1, 1))
        if transform is not None:
            _data = transform(df["Sunspot_Number"].to_numpy().reshape(-1, 1))
            self._sunspot = torch.tensor(_data)
        else:
            self._sunspot = torch.tensor(df["Sunspot_Number"].to_numpy())
        self.len = self._sunspot.shape[0]

    def __getitem__(self, index: int):
        return self._year[index], self._sunspot[index]

    def __len__(self):
        return self.len

def split_ts_data(data, p_test, window_size=1):
    """ Split time-series data

    Params
    ------
    data: DataFrame
        full dataset
    p_test: float
        data partition for test
    window_size: int
        size of data for each training batch

    Returns
    -------
    train_seq, test_seq: list
        splitted datasets
    """
    tot_seq = []
    for i in range(len(data) - window_size):
        in_seq = data[i:i+window_size]
        out_target = data[i+window_size]
        tot_seq.append((in_seq, out_target))
    n_tot = len(tot_seq)
    n_test = int(n_tot * p_test)
    n_train = n_tot - n_test
    train_seq = tot_seq[:n_train]
    test_seq = tot_seq[n_train:]
    return train_seq, test_seq

transform = MinMaxScaler(feature_range=(0, 1))

from torch import nn

torch.manual_seed(1)

class LSTM(nn.Module):
    def __init__(self, input_size=1, hidden_size=20, output_size=1):
        super(LSTM, self).__init__()
        self.hidden_size = hidden_size
        self.lstm = nn.LSTM(input_size, hidden_size, batch_first=True)
        self.linear = nn.Linear(hidden_size, output_size)

    def forward(self, x_seq):
        lstm_out, _ = self.lstm(x_seq)
        out = self.linear(lstm_out[:, -1, :])
        return out

model = LSTM(1, 50, 1).to(torch.float64)
print(model)
f_loss = nn.MSELoss()
optimizer = torch.optim.SGD(model.parameters(), lr=0.01, momentum=0.7)
